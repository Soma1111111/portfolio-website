# code, mostly

One scan in, an estimate out — a slice of the brain-age pipeline.

brain_age.py ◐ predicting 
# estimate age from a single fMRI scan
import torch

model = torch.load("swin_unet.pt", map_location="cpu")
scan = torch.load("subject_042.nii.gz")

with torch.no_grad():
    age = model(scan.unsqueeze(0)).item()

print(f"predicted age: {age:.1f} ± 3.2 years")
